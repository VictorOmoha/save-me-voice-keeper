name: Emulator Tests

# SAVE-005 (lane V3) — runs the Firebase emulator harness as a one-command path
# and uploads content-safe artifacts. The emulator suite, seed, and rules tests
# are owned by Atlas Backend; this workflow only *invokes* the runner
# (test/emulator/run.mjs) that the root package.json script wraps.
#
# Hard constraints honored here:
#   - No production credentials. The harness runs against a `demo-saveme`
#     project, which firebase-tools treats as fully local.
#   - No secrets in this file.
#   - Artifacts are content-safe: emulator logs + JUnit XML only. No seeded
#     data, no user content.

on:
  push:
    branches: ["main", "master"]
  pull_request:
  workflow_dispatch:

# Cancel superseded runs on the same ref to save CI minutes.
concurrency:
  group: emulator-tests-${{ github.ref }}
  cancel-in-progress: true

jobs:
  emulator:
    name: Firestore/Storage rules (emulator)
    runs-on: ubuntu-latest
    timeout-minutes: 20

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Setup Node
        uses: actions/setup-node@v4
        with:
          node-version: 20
          cache: npm

      # The Firestore emulator is a Java jar; it needs a JRE.
      - name: Setup Java (for Firestore emulator)
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: "21"

      - name: Install root dependencies
        run: npm ci

      - name: Install Functions dependencies
        run: npm --prefix functions ci

      - name: Install firebase-tools
        run: npm install --no-save firebase-tools

      # Pre-download emulator binaries so the runner doesn't time out fetching
      # them mid-run. Uses the local firebase-tools installed above.
      - name: Pre-fetch emulator binaries
        run: npx firebase setup:emulators:firestore && npx firebase setup:emulators:storage

      - name: Run emulator harness
        run: npm run test:emulator:ci
        env:
          # Belt-and-suspenders: the runner must refuse to run without these.
          SAVE_ME_EMULATOR: "1"
          GCLOUD_PROJECT: demo-saveme
          CI: "true"

      - name: Upload emulator JUnit results
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: emulator-junit
          path: test-results/emulator-junit.xml
          if-no-files-found: warn
          retention-days: 14

      - name: Upload emulator logs
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: emulator-logs
          path: |
            firebase-debug.log
            firestore-debug.log
            ui-debug.log
            emulator-*.log
          if-no-files-found: warn
          retention-days: 14
